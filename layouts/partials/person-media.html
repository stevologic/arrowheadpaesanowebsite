{{- /* Credited X photograph for a named person, or nothing. */ -}} {{- $name := .name | default "" -}} {{- $n := .n | default dict -}} {{- $bundle := index .site.Data "x-photo" | default dict -}} {{- $file := index $bundle "people" | default dict -}} {{- $people := index $file "people" | default dict -}} {{- $row := index $people ($name | urlize) -}} {{- if not $row -}} {{- $xp := $n.xPhoto | default dict -}} {{- $askedId := .postId | default "" -}} {{- if and $name $askedId $xp.postId (eq $xp.postId $askedId) -}} {{- $row = $xp -}} {{- else if and $name $xp.alt (ge (len $name) 5) (in ($xp.alt | lower) ($name | lower)) -}} {{- $row = $xp -}} {{- end -}} {{- end -}} {{- $asset := $row.asset | default "" -}} {{- $handle := $row.handle | default "" -}} {{- $postId := $row.postId | default "" -}} {{- $postUrl := $row.postUrl | default "" -}} {{- $ok := and $name $postId $handle $postUrl $asset (findRE `^images/x-photo/[A-Za-z0-9._-]+\.(?:webp|png|jpe?g)$` $asset) (not (in $asset "..")) (findRE `^https://x\.com/[A-Za-z0-9_]+/status/[0-9]+$` $postUrl) (fileExists (printf "public/%s" $asset)) -}} {{- if $ok -}} Photo: @{{ $handle }} on X {{- end -}}
